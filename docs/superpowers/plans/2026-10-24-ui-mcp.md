# UI + MCP (Sub-project 6) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Every job (Ask, Checklist, Crosswalk, Gap check) usable from an MCP client, and a UI a first-time visitor understands: a name, a Home page, an Evaluation page about the compliance evals, fixed rough edges and a polish pass.

**Architecture:** The MCP server (`backend/app/mcp_server.py`) wraps the same pure services the REST API uses (`build_checklist`, `crosswalk.related`, `gapcheck.plan/check_unit`, `parse_clauses`), validating input with the API's own schemas. The UI reads eval snapshots that `eval/publish_results.py` writes into `eval/results/`; all shaping is pure functions in `frontend/services/eval_data.py` and `presenters.py`, unit-tested without Streamlit.

**Tech Stack:** mcp 2.3.0 (`MCPServer`, in-memory `Client`), FastAPI, Streamlit 1.66.0, Altair, pandas, pytest.

## Global Constraints

- Spec: `docs/superpowers/specs/2026-10-20-compliance-copilot-design.md` §17.
- All MCP tools read-only (`_READ_ONLY` annotations); nothing stored, cached or logged; the MCP path never uses the response cache.
- MCP server id stays `rag-agent-workbench`; display text says "Compliance Workbench".
- Limits mirror the API schemas: policy text 200,000 chars and ≤ 12 articles (`GapCheckRequest`), standard copy 1,000,000 chars (`ParseRequest`); gap-check budget 12 units / 40,000 estimated tokens (`gapcheck.MAX_UNITS`, `MAX_TOKENS`).
- Invariants unchanged: bounded loop, deterministic abstention, embedder-independent faithfulness checks, injection hardening, quote verification ("unverified" marking).
- No Pinecone writes. No push, no deploy. Never read or print `.env` values. Never commit `docs/INTERVIEW_MASTERFILE.md`, `docs/RAG_INTERVIEW_MASTERFILE.md` or `eval/labels/`.
- Copy: sentence case, buttons say what they do, "not legal advice" where answers are shown.
- Commands (Windows, Git Bash, repo root):
  - backend suite: `.venv-relock/Scripts/python -m pytest -p no:cacheprovider tests -q`
  - UI smoke: `PYTHONPATH=frontend .venv-revive-ui/Scripts/python tests/ui/smoke_app.py`
  - API: `API_KEY=local-dev-key .venv-relock/Scripts/python -m uvicorn app.main:app --app-dir backend --port 8767`
  - UI: `BACKEND_BASE_URL=http://127.0.0.1:8767 API_KEY=local-dev-key .venv-revive-ui/Scripts/python -m streamlit run frontend/app.py --server.headless true --server.port 8766`

Verified SDK behaviour this plan relies on (probed 2026-10-24 against the installed mcp 2.3.0):
a tool returning `Dict[str, Any]` puts it under `structured_content["result"]`; a tool returning a pydantic model puts its fields at the top level; argument validation errors come back as `is_error` with the pydantic message (allowed literal values listed, "Extra inputs are not permitted"); `Context.report_progress(i, n, msg)` reaches `call_tool(..., progress_callback=cb)` as `cb(i, n, msg)`; `Tool.annotations.read_only_hint`; list-returning JSON resources are serialised as JSON text.

```python
@mcp.tool(annotations=_READ_ONLY)
async def checklist(profile: Optional[StrictProfile] = None, as_of: Optional[str] = None,
                    with_text: bool = False) -> Dict[str, Any]:
    """Which articles of the EU AI Act, GDPR, NIS2, DORA and the Cyber Resilience Act bind a profile on as_of.

    Deterministic (a rules table, no model). rag://facts lists the facts and their allowed values; include only
    facts the user stated. Returns groups now / coming / depends, the questions that would decide `depends`,
    voluntary NIST practices and what does not apply. with_text adds each paragraph's official text.
    """
    d = _date(as_of)
    prof = profile.model_dump(exclude_none=True) if profile else {}
    return await anyio.to_thread.run_sync(lambda: build_checklist(prof, d, with_text=with_text))


@mcp.tool(annotations=_READ_ONLY)
async def crosswalk(provision: str, hops: int = 2, official_only: bool = False, per_group: int = 4) -> Dict[str, Any]:
    """Requirements in other frameworks that published mappings link to a provision.

    provision: an id such as 'nist-csf/GV.SC-01' or 'gdpr/art-32/par-1', or its exact ref such as
    'GDPR Art. 32(1)'. Related requirements, not equivalence; ISO/IEC matches are clause ids only.
    """
    if hops not in (1, 2) or not 1 <= per_group <= 20:
        raise ToolError("hops must be 1 or 2 and per_group between 1 and 20")
    corpus, _ = regs_corpus()
    pid = provision.strip()
    if pid not in corpus and pid not in xw.graph():
        pid = _ref_ids().get(pid.lower(), "")
    if not pid:
        raise ToolError(f"Unknown provision: {provision!r}. Use an id such as 'gdpr/art-32/par-1' or its exact ref.")
    found = xw.related(pid, hops=hops, official_only=official_only, per_group=per_group)
    return {"provision": pid, "ref": corpus[pid]["ref"] if pid in corpus else pid, **found, "caveat": xw.CAVEAT}
```

```python
@mcp.resource("rag://facts", mime_type="application/json")
def facts() -> List[Dict[str, Any]]:
    """Profile facts for `checklist` and `gap_check`: id, question, type, allowed values and their labels."""
    keys = ("id", "type", "group", "label", "values", "labels", "help")
    return [{k: f[k] for k in keys if k in f} for f in load_facts()]


@mcp.resource("rag://obligations", mime_type="application/json")
def obligations() -> Dict[str, Any]:
    """The checklist's rules: each duty-bearing article and the condition over rag://facts that makes it apply."""
    catalog, index = load_catalog(), article_index()
    rows = []
    for ob in catalog["obligations"]:
        head = index[ob["article"]][0]
        rows.append({"article": ob["article"], "ref": _article_ref(head), "title": head.get("title") or "",
                     "framework": head["framework"], "kind": ob.get("kind", "obligation"), "when": ob["when"],
                     **({"include": ob["include"]} if ob.get("include") else {})})
    return {"defs": catalog["defs"], "obligations": rows}


@mcp.resource("rag://crosswalk-sources", mime_type="application/json")
def crosswalk_sources() -> List[Dict[str, Any]]:
    """The published mappings behind `crosswalk`: title, publisher, official or community, reuse terms, caveat."""
    keys = ("id", "title", "publisher", "kind", "url", "reuse", "caveat")
    return [{k: s[k] for k in keys if k in s} for s in xw.sources().values()]
```

```python
class CopyText(BaseModel):
    text: str = Field(..., min_length=1, max_length=1_000_000, description="Your licensed copy's text.")
    name: Optional[Literal["iso-27001", "iso-42001"]] = Field(None, description="Override detection.")


def _copies(standards: Optional[List[CopyText]]) -> List[Dict[str, Any]]:
    """Parse attached licensed copies for this call only (spec §17.1); never cached or logged."""
    out = []
    for c in standards or []:
        name = c.name or detect_standard(c.text)
        if not name:
            raise ToolError("Could not tell which standard this is. Pass name 'iso-27001' (ISO/IEC 27001) or "
                            "'iso-42001' (ISO/IEC 42001).")
        clauses = parse_clauses(c.text, name)
        if not clauses:
            raise ToolError(f"No numbered clauses found in this copy of {STANDARDS[name]}.")
        out.append({"name": name, "clauses": clauses})
    return out
```

```python
@mcp.tool(annotations=_READ_ONLY)
async def gap_check(policy_text: str, ctx: Context, articles: Optional[List[str]] = None,
                    profile: Optional[StrictProfile] = None, as_of: Optional[str] = None) -> Dict[str, Any]:
    """Check a policy, duty by duty, against articles or annex sections, with verbatim quotes for every verdict.

    articles: keys such as 'gdpr/art-33' or 'nis2-ir/anx-1#11'; without them, the articles the profile's
    checklist says apply now or later. At most 12 per run and about 40,000 tokens; progress is reported per
    article. 'unverified' means a quote was not found verbatim and needs review. Nothing is stored.
    """
    try:
        req = GapCheckRequest(policy_text=policy_text, articles=articles, profile=profile, as_of=as_of)
    except ValidationError as exc:
        raise ToolError(str(exc)) from exc
    prof = req.profile.model_dump(exclude_none=True) if req.profile else {}
    try:
        p = gapcheck.plan(req.policy_text, req.articles, prof, req.as_of or date.today())
    except ValueError as exc:
        raise ToolError(str(exc)) from exc
    msg = gapcheck.budget_error(p)
    if msg:
        raise ToolError(msg)
    llm, results, n = get_llm(), [], len(p["units"])
    for i, unit in enumerate(p["units"], 1):
        results.append(await anyio.to_thread.run_sync(gapcheck.check_unit, unit, llm))
        await ctx.report_progress(i, n, unit["ref"])
    return {"plan": gapcheck.plan_summary(p), "results": results, "summary": gapcheck.summarize(results)}
```

```python
#!/usr/bin/env python3
"""Snapshot the latest eval reports into eval/results/ (committed) for the UI's Evaluation and Home pages.

eval/reports/ is gitignored scratch; eval/results/ is what the deployed app reads.
Usage: python eval/publish_results.py
"""
import json
import re
import shutil
from collections import Counter
from pathlib import Path
from typing import List, Optional

REPORTS = Path(__file__).resolve().parent / "reports"
RESULTS = REPORTS.parent / "results"
REGS_DATA = REPORTS.parents[1] / "backend" / "app" / "data" / "regs"
STAMP = r"\d{8}T\d{6}Z\.json$"
KINDS = {
    "retrieval": re.compile(rf"^{STAMP}"),
    "sweep": re.compile(rf"^sweep_{STAMP}"),
    "rerank_ab": re.compile(rf"^ab_{STAMP}"),
    "answers": re.compile(rf"^answers_{STAMP}"),
    "adversarial": re.compile(rf"^adversarial_{STAMP}"),
    "regs_abstain": re.compile(rf"^regs_abstain_{STAMP}"),
    "router": re.compile(rf"^router_{STAMP}"),
}
MODES = ["lexical", "dense", "fused", "diversified", "diversified-boost", "diversified-inject"]
REGS_K = 6  # provisions sent to the model on the compliance path (REGS_TOP_K)
INSTRUMENTS = {"eu-ai-act": "EU AI Act", "gdpr": "GDPR", "nis2": "NIS2",
               "nis2-ir": "NIS2 Implementing Regulation 2024/2690", "dora": "DORA", "cra": "Cyber Resilience Act",
               "nist-800-53": "NIST SP 800-53", "nist-csf": "NIST CSF 2.0", "nist-ai-rmf": "NIST AI RMF"}


def _load(p: Path) -> dict:
    return json.loads(p.read_text(encoding="utf-8"))


def _runs(reports: Path, prefix: str) -> List[Path]:
    pat = re.compile(rf"^{re.escape(prefix)}_{STAMP}")
    return sorted(p for p in reports.glob("*.json") if pat.match(p.name))


def _at_k(reports: Path, mode: str) -> List[dict]:
    return [r for r in map(_load, _runs(reports, f"regs_{mode}")) if r.get("k") == REGS_K]


def regs_retrieval(reports: Path) -> Optional[dict]:
    """The latest production-mode (diversified) retrieval run at k = 6, with per-type and per-question rows."""
    runs = _at_k(reports, "diversified")
    return runs[-1] if runs else None


def regs_modes(reports: Path) -> dict:
    """Latest run at k = 6 per retrieval mode, for the comparison chart."""
    out = {}
    for mode in MODES:
        runs = _at_k(reports, mode)
        if runs:
            agg = runs[-1]["aggregate"]
            out[mode] = {"n": runs[-1]["n"], **{m: agg[m] for m in ("recall", "mrr", "ndcg") if m in agg}}
    return out


def gapcheck_runs(reports: Path) -> Optional[dict]:
    """The latest golden gap check, plus the headline numbers of every run of the same prompt (the judge is a
    live model, so one run is not the whole story)."""
    paths = _runs(reports, "gapcheck")
    if not paths:
        return None
    latest, runs = _load(paths[-1]), []
    for p in paths:
        r = _load(p)
        if r.get("prompt_sha") == latest.get("prompt_sha"):
            runs.append({"run": p.stem.split("_", 1)[1],
                         **{k: r[k] for k in ("accuracy", "verified_rate", "missing_caught", "missing_planted", "tokens")},
                         "false_covered": len(r["false_covered"])})
    return {**latest, "runs": runs}


def corpus_summary(data_dir: Path) -> List[dict]:
    """Per instrument: binding, provision count, official source and retrieval date (Home page)."""
    manifest = {m["key"]: m for m in _load(data_dir / "manifest.json")}
    counts, binding = Counter(), {}
    for line in (data_dir / "regs-v1.jsonl").read_text(encoding="utf-8").splitlines():
        if line.strip():
            p = json.loads(line)
            key = p["id"].split("/", 1)[0]
            counts[key] += 1
            binding[key] = p["binding"]
    return [{"key": k, "name": name, "binding": binding[k], "provisions": counts[k], "url": manifest[k]["url"],
             "retrieved": manifest[k]["retrieved"]} for k, name in INSTRUMENTS.items() if k in counts]


def publish(reports: Path = REPORTS, results: Path = RESULTS, data_dir: Path = REGS_DATA) -> List[str]:
    results.mkdir(exist_ok=True)
    done = []
    for kind, pattern in KINDS.items():
        candidates = sorted(p for p in reports.glob("*.json") if pattern.match(p.name))
        if kind == "rerank_ab":  # the multi-k A/B is the one the design decision cites
            candidates = [p for p in candidates if _load(p).get("multi_k")]
        if candidates:
            shutil.copyfile(candidates[-1], results / f"{kind}.json")
            done.append(kind)
    built = {"regs_retrieval": regs_retrieval(reports), "regs_modes": regs_modes(reports) or None,
             "gapcheck": gapcheck_runs(reports), "corpus": corpus_summary(data_dir)}
    for kind, data in built.items():
        if data:
            (results / f"{kind}.json").write_text(json.dumps(data, indent=2), encoding="utf-8")
            done.append(kind)
    return done


if __name__ == "__main__":
    print("published:", ", ".join(publish()))
```

```python
MODE_LABELS = {"lexical": "Lexical (BM25)", "dense": "Dense", "fused": "Fused (RRF)",
               "diversified": "Diversified (production)", "diversified-boost": "+ crosswalk boost",
               "diversified-inject": "+ crosswalk inject"}
_TYPES = {"concept": "About a topic", "cross": "Across two frameworks", "reference": "Naming a provision"}
_ROUTES = ["ask", "checklist", "gap_check", "clarify"]
_BINDING = {"regulation": "Binding regulation", "directive": "Binding directive (through national law)",
            "voluntary_framework": "Voluntary framework"}


def mode_rows(modes: Dict[str, Any]) -> List[Dict[str, Any]]:
    """One bar per retrieval mode, in a fixed order (never sorted by value); production is marked."""
    return [{"mode": label, "recall": round(modes[k]["recall"], 3), "mrr": round(modes[k]["mrr"], 3),
             "n": modes[k]["n"], "production": k == "diversified"} for k, label in MODE_LABELS.items() if k in modes]


def type_rows(retrieval: Dict[str, Any]) -> List[Dict[str, Any]]:
    k = retrieval["k"]
    return [{"Questions": _TYPES.get(t, t), "n": v["n"], f"Recall@{k}": round(v["recall"], 3),
             "MRR": round(v["mrr"], 3), f"nDCG@{k}": round(v["ndcg"], 3)}
            for t, v in retrieval["aggregate"]["by_type"].items()]


def confusion_rows(confusion: Dict[str, int]) -> List[Dict[str, Any]]:
    """{'gap_check->ask': 2} -> one row per request type, one column per job it was routed to."""
    seen = {part for key in confusion for part in key.split("->")}
    labels = [r for r in _ROUTES if r in seen] + sorted(seen - set(_ROUTES))
    table = {a: {b: 0 for b in labels} for a in labels}
    for key, n in confusion.items():
        a, b = key.split("->")
        table[a][b] = n
    return [{"Request": a, **table[a]} for a in labels]


def _count(x: Any) -> int:
    return x if isinstance(x, int) else len(x)


def gap_range(gap: Dict[str, Any]) -> Dict[str, Any]:
    runs = gap.get("runs") or [gap]
    acc, ver = [r["accuracy"] for r in runs], [r["verified_rate"] for r in runs]
    return {"runs": len(runs), "accuracy": (min(acc), max(acc)), "verified_rate": (min(ver), max(ver)),
            "false_covered": max(_count(r["false_covered"]) for r in runs)}


def span(pair: Tuple[float, float], pct: bool = False) -> str:
    fmt = (lambda v: f"{v:.0%}") if pct else (lambda v: f"{v:.2f}")
    lo, hi = fmt(pair[0]), fmt(pair[1])
    return lo if lo == hi else f"{lo}–{hi}"


def headline(results: Dict[str, Any]) -> List[Dict[str, str]]:
    """Home page: the measured numbers that matter most, each with where it comes from."""
    out = []
    r = results.get("regs_retrieval")
    if r:
        out.append({"label": f"Right provision in the top {r['k']}", "value": f"{r['aggregate']['recall']:.0%}",
                    "note": f"Recall@{r['k']} on {r['n']} golden questions"})

    if a:
        out.append({"label": "Out-of-scope questions declined", "value": f"{a['correct_abstain']} of {a['unanswerable']}",
                    "note": f"{a['false_abstain']} of {a['answerable']} in-scope questions wrongly declined"})
    rt = results.get("router")
    if rt:
        out.append({"label": "Requests routed to the right job", "value": f"{rt['accuracy']:.0%}",
                    "note": f"{rt['n']} labelled requests"})
    g = results.get("gapcheck")
    if g:
        rng = gap_range(g)
        out.append({"label": "Gap check: planted gaps called covered", "value": str(rng["false_covered"]),
                    "note": f"Across {rng['runs']} runs; quotes verified {span(rng['verified_rate'], pct=True)}"})
    return out
```

```python
def crosswalk_compact(xw: Optional[Dict[str, Any]], standards: Optional[Dict[str, Any]] = None,
                      inline: int = 3) -> List[Dict[str, Any]]:
    """Per group: the first `inline` refs as badges on one line, their mapping sources, and the remaining lines."""
    out = []
    for group, lines, hidden in crosswalk_lines(xw, standards):
        top = lines[:inline]
        out.append({"group": group,
                    "line": " ".join(f":{color}-badge[{esc(label.split(' · ')[0])}]" for label, color, _ in top),
                    "notes": " · ".join(dict.fromkeys(note for _, _, note in top)),
                    "rest": lines[inline:], "hidden": hidden})
    return out
```

```python
"""Live MCP check over Streamable HTTP (spec §17.6): python scripts/smoke_mcp.py [url]
Needs the API running locally and API_KEY in the environment (never printed). Spends a few thousand Groq tokens;
no Pinecone writes."""
import os
import sys
from pathlib import Path

import anyio
import httpx2
from mcp.client.client import Client
from mcp.client.streamable_http import streamable_http_client

sys.path.insert(0, str(Path(__file__).resolve().parents[1]))
from tests.regs.test_standards import SYNTH  # noqa: E402  (synthetic copy: real clause ids, invented wording)

POLICY = ("Supplier security policy\n\nWe assess the security practices of every supplier before signing a contract "
          "and review critical suppliers once a year.\n\nContracts include security requirements and a right to audit.")
PROFILE = {"ai_system": True, "ai_roles": ["deployer"], "eu_established": True, "high_risk": "annex_iii"}


async def main(url: str) -> None:
    http = httpx2.AsyncClient(headers={"X-API-Key": os.environ["API_KEY"]}, timeout=180)
    async with Client(streamable_http_client(url, http_client=http), read_timeout_seconds=300) as c:
        print("tools:", sorted(t.name for t in (await c.list_tools()).tools))
        print("rag://facts:", len((await c.read_resource("rag://facts")).contents[0].text), "chars")
        cl = (await c.call_tool("checklist", {"profile": PROFILE})).structured_content["result"]
        print("checklist counts:", cl["counts"])
        xw = (await c.call_tool("crosswalk", {"provision": "nist-csf/GV.SC-01"})).structured_content["result"]
        print("crosswalk:", len(xw["items"]), "matches, first", [m["ref"] for m in xw["items"][:3]])
        seen = []

        async def on_progress(progress, total, message):
            seen.append(f"{progress:g}/{total:g} {message}")

        gap = (await c.call_tool("gap_check", {"policy_text": POLICY, "articles": ["nis2-ir/anx-1#5"]},
                                 progress_callback=on_progress)).structured_content["result"]
        print("gap_check progress:", seen, "summary:", gap["summary"])
        ans = (await c.call_tool("ask", {"question": "What does ISO/IEC 27001 A.5.19 require?",
                                         "standards": [{"text": SYNTH}]})).structured_content
        print("ask with your copy: abstained", ans["abstained"], "sources", [s["ref"] or s["title"] for s in ans["sources"]][:4])


if __name__ == "__main__":
    anyio.run(main, sys.argv[1] if len(sys.argv) > 1 else "http://127.0.0.1:8767/mcp")
```
